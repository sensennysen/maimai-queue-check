Create a new migration file `supabase/migrations/20260320141500_finish_game.sql` containing the `finish_game` function:
```sql
CREATE OR REPLACE FUNCTION public.finish_game(
  p_current_playing_id UUID DEFAULT NULL,
  p_next_waiting_id UUID DEFAULT NULL
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  transition_time TIMESTAMP WITH TIME ZONE := NOW();
BEGIN
  IF p_current_playing_id IS NOT NULL THEN
    UPDATE queue_entries
    SET status = 'completed',
        ended_at = transition_time
    WHERE id = p_current_playing_id;
  END IF;

  IF p_next_waiting_id IS NOT NULL THEN
    UPDATE queue_entries
    SET status = 'playing',
        started_at = transition_time
    WHERE id = p_next_waiting_id;
  END IF;
END;
$$;
```
</action>
